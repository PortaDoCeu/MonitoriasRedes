% ── Fundo: céu + ondas + reflexo do sol ───────────────────────────────────
%
%   Filosofia: o fundo não é decoração, é atmosfera.
%   Três camadas sobrepostas, todas suaves:
%     1. Reflexo solar no canto superior direito (coral + apricot)
%     2. Mar silencioso no canto inferior esquerdo (cerulean → teal)
%     3. Linhas de onda mínimas — o vento passando raso
%
\setbeamertemplate{background}{
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (\paperwidth,\paperheight);

    % Base: areia / luz amarela
    \fill[LightYellow] (0,0) rectangle (\paperwidth,\paperheight);

    % ── Reflexo do sol: mancha orgânica, canto superior direito ──
    \fill[SoftApricot, opacity=0.28]
      (\paperwidth, \paperheight)
      .. controls (\paperwidth-3.2, \paperheight+0.4) and (\paperwidth-1.0, \paperheight-2.8) ..
      (\paperwidth-0.4, \paperheight-4.5)
      .. controls (\paperwidth+0.6, \paperheight-3.0) and (\paperwidth+0.2, \paperheight-0.8) ..
      cycle;

    \fill[VibrantCoral, opacity=0.10]
      (\paperwidth, \paperheight)
      .. controls (\paperwidth-2.0, \paperheight+0.2) and (\paperwidth-0.5, \paperheight-1.8) ..
      (\paperwidth-0.3, \paperheight-3.0)
      .. controls (\paperwidth+0.4, \paperheight-2.0) and (\paperwidth+0.1, \paperheight-0.5) ..
      cycle;

    % ── Mar: mancha fluida, canto inferior esquerdo ──
    \fill[Cerulean, opacity=0.16]
      (0,0)
      .. controls (0.5, 1.8) and (2.2, 0.6) ..
      (4.8, 0.8)
      .. controls (3.5, 2.2) and (1.2, 3.0) ..
      (0, 3.8)
      -- cycle;

    \fill[TropicalTeal, opacity=0.13]
      (0,0)
      .. controls (0.4, 1.2) and (1.8, 0.4) ..
      (3.5, 0.5)
      .. controls (2.4, 1.6) and (0.8, 2.2) ..
      (0, 2.8)
      -- cycle;

    % ── Ondas: traços curvos levíssimos, como vento raso ──
    % (aparecendo na faixa inferior — onde está o "mar")
    \draw[TropicalTeal, opacity=0.20, line width=0.5pt, rounded corners=8pt]
      (0.3, 1.05) .. controls (0.9, 0.92) .. (1.6, 1.05);
    \draw[TropicalTeal, opacity=0.16, line width=0.4pt, rounded corners=8pt]
      (0.6, 0.72) .. controls (1.4, 0.58) .. (2.2, 0.72);
    \draw[Cerulean, opacity=0.14, line width=0.4pt, rounded corners=8pt]
      (1.0, 1.38) .. controls (1.9, 1.22) .. (2.8, 1.38);
    \draw[TropicalTeal, opacity=0.12, line width=0.35pt, rounded corners=8pt]
      (0.2, 1.75) .. controls (1.2, 1.60) .. (2.0, 1.75);

    % ── Linha do horizonte: finíssima, quase não existe ──
    \draw[TropicalTeal, opacity=0.12, line width=0.3pt]
      (0, 2.90) .. controls (4, 2.78) and (8, 2.92) .. (\paperwidth, 2.85);

  \end{tikzpicture}
}

% ── Bloco customizado: "onda" ──────────────────────────────────────────────
%   Use \begin{waveblock}{Título} ... \end{waveblock}
%   para um bloco com borda esquerda fluida e fundo apricot
\newenvironment{waveblock}[1]{%
  \begin{tikzpicture}
    \node[
      fill=SoftApricot!45,
      draw=TropicalTeal,
      line width=0.4pt,
      rounded corners=6pt,
      inner sep=10pt,
      text width=\linewidth-24pt
    ] \bgroup
      \textbf{\color{Cerulean}#1}\\[4pt]
      \color{DeepSea}
}{%
    \egroup;
  \end{tikzpicture}
}

% ── Tag de categoria ──────────────────────────────────────────────────────
%   \wavetag{TropicalTeal}{Conceito}   \wavetag{VibrantCoral}{Destaque}
\newcommand{\wavetag}[2]{%
  \tikz[baseline=(t.base)]\node[
    fill=#1,
    text=LightYellow,
    rounded corners=8pt,
    inner xsep=7pt,
    inner ysep=2.5pt,
    font=\scriptsize\bfseries
  ] (t) {#2};%
}
